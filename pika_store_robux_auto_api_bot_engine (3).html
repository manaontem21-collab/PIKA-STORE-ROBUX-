Kode Backend Webhook & Bot API (Node.js)

Memverifikasi callback Webhook pembayaran lalu mengeksekusi pengiriman via Username.

// server.js - Pika Store Robux Webhook & Bot Delivery API (Owner Only)
const express = require('express');
const crypto = require('crypto');
const noblox = require('noblox.js');
const app = express();
app.use(express.json());

// Endpoint API Pengecekan Username -> Display Name & Avatar Roblox
app.get('/v1/roblox/check-username/:username', async (req, res) => {
    try {
        const cleanUsername = String(req.params.username || '').replace(/^@+/, '').trim();
        if (!cleanUsername) {
            return res.status(400).json({ success: false, message: 'Username wajib diisi' });
        }

        const userLookupRes = await fetch('https://users.roblox.com/v1/usernames/users', {
            method: 'POST',
            headers: { 'Content-Type': 'application/json' },
            body: JSON.stringify({ usernames: [cleanUsername], excludeBannedUsers: true })
        });
        const userLookupData = await userLookupRes.json();
        const user = userLookupData.data && userLookupData.data[0];

        if (!user) {
            return res.status(404).json({ success: false, message: 'Username Roblox tidak ditemukan' });
        }

        const thumbUrl = `https://thumbnails.roblox.com/v1/users/avatar-headshot?userIds=${user.id}&size=150x150&format=Png&isCircular=false`;
        const thumbRes = await fetch(thumbUrl);
        const thumbData = await thumbRes.json();
        const avatarUrl = (thumbData.data && thumbData.data[0] && thumbData.data[0].imageUrl) || '';

        return res.json({
            success: true,
            userId: user.id,
            username: user.name,
            displayName: user.displayName,
            avatarUrl
        });
    } catch (err) {
        return res.status(500).json({ success: false, error: err.message });
    }
});

const botAccounts = [
    { id: 'BOT_ROBUX_SG_01', cookie: process.env.BOT_COOKIE_1, balance: 68500 },
    { id: 'BOT_ROBUX_ID_02', cookie: process.env.BOT_COOKIE_2, balance: 49350 }
];

async function pickAvailableBot(requiredRobux) {
    const bot = botAccounts.find((b) => b.balance >= requiredRobux && b.cookie);
    if (!bot) throw new Error('Stok Robux pada seluruh akun bot tidak mencukupi');
    await noblox.setCookie(bot.cookie);
    return bot;
}

async function sendRobuxViaUsername({ username, robuxAmount }) {
    const userId = await noblox.getIdFromUsername(username);
    const bot = await pickAvailableBot(robuxAmount);
    const transferReceipt = await noblox.sendRobuxByUsername({ targetUserId: userId, amount: robuxAmount });
    bot.balance -= robuxAmount;
    return { userId, username, botId: bot.id, robuxAmount, transferReceipt };
}

app.post('/v1/webhook/payment', async (req, res) => {
    try {
        const signature = req.headers['x-callback-signature'];
        const computed = crypto
            .createHmac('sha256', process.env.PAYMENT_PRIVATE_KEY || 'dev_secret')
            .update(JSON.stringify(req.body))
            .digest('hex');

        if (signature !== computed) {
            return res.status(403).json({ success: false, message: 'Invalid signature' });
        }

        const { merchant_ref, status, username, robux_amount } = req.body;
        if (status === 'PAID') {
            const delivery = await sendRobuxViaUsername({
                username,
                robuxAmount: robux_amount
            });
            return res.status(200).json({ success: true, orderId: merchant_ref, delivery });
        }
        return res.status(200).json({ success: true, status });
    } catch (err) {
        return res.status(500).json({ success: false, error: err.message });
    }
});

app.listen(3000, () => console.log('Pika Store Webhook & Bot API berjalan di port 3000'));
